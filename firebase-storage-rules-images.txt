rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Dhamma post images - only authenticated users can upload
    match /dhamma_posts/{imageFile} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Audio files - only authenticated users can upload
    match /audio/{audioFile} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Default rule - deny all other access
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
